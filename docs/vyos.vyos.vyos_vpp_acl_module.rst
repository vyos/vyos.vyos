vyos.vyos.vyos_vpp_acl

**VPP dataplane ACL resource module**


Version added: 7.0.0

.. contents::
   :local:
   :depth: 1


Synopsis
--------
- This module manages the access control lists of the accelerated dataplane (VPP) on devices running VyOS - IP and MAC ACLs, and their bindings to dataplane interfaces.

Notes
-----

.. note::
   - Tested against VyOS 1.5.1.
   - This module works with connection ``network_cli``.
   - ACLs can only be bound to interfaces that are part of the dataplane (``vpp settings interface``).
   - IP and MAC ACLs are handled independently. With *state=replaced*, the listed ACLs and interface bindings are replaced as given; with *state=overridden*, ACLs and bindings that are not listed are also deleted, bindings before ACLs.
   - When not configured, VyOS uses the protocol ``all`` for IP rules and the mask ``ff:ff:ff:ff:ff:ff`` for MAC rules; the module reports only configured values.
   - For more information on using Ansible to manage network devices see the :ref:`Ansible Network Guide <network_guide>`



Examples
--------

.. code-block:: yaml

    # Using merged

    # Before state:
    # -------------
    #
    # vyos@vyos:~$ show configuration commands | grep 'vpp acl'
    #

    - name: Permit HTTPS from partners into eth1
      vyos.vyos.vyos_vpp_acl:
        config:
          ip:
            acls:
              - name: WEB-IN
                description: Inbound web traffic
                rules:
                  - number: 10
                    action: permit
                    protocol: tcp
                    source:
                      prefix: 192.0.2.0/24
                    destination:
                      prefix: 198.51.100.10/32
                      port: "443"
                  - number: 20
                    action: deny
            interfaces:
              - name: eth1
                input:
                  - acl_tag: 1
                    tag_name: WEB-IN
        state: merged

    # Task output
    # -----------
    #
    #    commands:
    #      - set vpp acl ip tag-name WEB-IN description 'Inbound web traffic'
    #      - set vpp acl ip tag-name WEB-IN rule 10 action permit
    #      - set vpp acl ip tag-name WEB-IN rule 10 protocol tcp
    #      - set vpp acl ip tag-name WEB-IN rule 10 source prefix 192.0.2.0/24
    #      - set vpp acl ip tag-name WEB-IN rule 10 destination prefix 198.51.100.10/32
    #      - set vpp acl ip tag-name WEB-IN rule 10 destination port 443
    #      - set vpp acl ip tag-name WEB-IN rule 20 action deny
    #      - set vpp acl ip interface eth1 input acl-tag 1 tag-name WEB-IN

    # Using merged, MAC ACL

    - name: Permit one vendor's MAC addresses on eth1
      vyos.vyos.vyos_vpp_acl:
        config:
          mac:
            acls:
              - name: MAC-FILTER
                rules:
                  - number: 10
                    action: permit
                    mac_address: 00:11:22:00:00:00
                    mac_mask: ff:ff:ff:00:00:00
            interfaces:
              - name: eth1
                tag_name: MAC-FILTER
        state: merged

    # Using replaced

    # Before state:
    # -------------
    #
    # set vpp acl ip tag-name WEB-IN rule 10 action 'permit'
    # set vpp acl ip tag-name WEB-IN rule 10 protocol 'tcp'
    # set vpp acl ip tag-name WEB-IN rule 20 action 'deny'

    - name: Replace the rules of WEB-IN
      vyos.vyos.vyos_vpp_acl:
        config:
          ip:
            acls:
              - name: WEB-IN
                rules:
                  - number: 10
                    action: permit
        state: replaced

    # Task output
    # -----------
    #
    #    commands:
    #      - delete vpp acl ip tag-name WEB-IN rule 10 protocol
    #      - delete vpp acl ip tag-name WEB-IN rule 20

    # Using overridden

    - name: Keep only the WEB-IN ACL and its binding
      vyos.vyos.vyos_vpp_acl:
        config:
          ip:
            acls:
              - name: WEB-IN
                rules:
                  - number: 10
                    action: permit
            interfaces:
              - name: eth1
                input:
                  - acl_tag: 1
                    tag_name: WEB-IN
        state: overridden

    # Using deleted

    - name: Remove the ACL bindings of eth1 and the WEB-IN ACL
      vyos.vyos.vyos_vpp_acl:
        config:
          ip:
            acls:
              - name: WEB-IN
            interfaces:
              - name: eth1
        state: deleted

    - name: Delete all dataplane ACL configuration
      vyos.vyos.vyos_vpp_acl:
        state: deleted

    # Task output (all ACL configuration)
    # -----------------------------------
    #
    #    commands:
    #      - delete vpp acl

    # Using gathered

    - name: Gather the dataplane ACL configuration
      vyos.vyos.vyos_vpp_acl:
        state: gathered

    # Using rendered

    - name: Render the commands for a MAC ACL
      vyos.vyos.vyos_vpp_acl:
        config:
          mac:
            acls:
              - name: MAC-FILTER
                rules:
                  - number: 10
                    action: deny
                    mac_address: 00:11:22:33:44:55
        state: rendered

    # Task output
    # -----------
    #
    #    rendered:
    #      - set vpp acl mac tag-name MAC-FILTER rule 10 action deny
    #      - set vpp acl mac tag-name MAC-FILTER rule 10 mac-address 00:11:22:33:44:55

    # Using parsed

    - name: Parse dataplane ACL configuration from device output
      vyos.vyos.vyos_vpp_acl:
        running_config: |
          set vpp acl ip tag-name WEB-IN rule 10 action 'permit'
          set vpp acl ip tag-name WEB-IN rule 10 tcp-flags is-set 'syn'
        state: parsed

    # Task output
    # -----------
    #
    #    parsed:
    #      ip:
    #        acls:
    #          - name: WEB-IN
    #            rules:
    #              - number: 10
    #                action: permit
    #                tcp_flags:
    #                  is_set:
    #                    - syn
